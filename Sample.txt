def encrypt(text, key):
    result = ""
    
    for ch in text:
        if ch.isupper():
            result += chr((ord(ch) - 65 + key) % 26 + 65)
        elif ch.islower():
            result += chr((ord(ch) - 97 + key) % 26 + 97)
        else:
            result += ch
            
    return result
    
def decrypt(text, key):
    return encrypt(text, -key)

def atbash(text):
    result = ""
    
    for ch in text:
        if ch.isupper():
            result+= chr(90 - (ord(ch) - 65))
        elif ch.islower():
            result += chr(122 - (ord(ch) - 97))
        else:
            result+=ch
            
    return result

10. Rail Fence


def encrypt(text, rails):
    fence = [["" for i in range(len(text))]
             for j in range(rails)]

    row = 0
    direction = 1

    for col in range(len(text)):
        fence[row][col] = text[col]

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    result = ""

    for i in range(rails):
        for j in range(len(text)):
            result += fence[i][j]

    return result


def decrypt(cipher, rails):
    fence = [["" for i in range(len(cipher))]
             for j in range(rails)]

    row = 0
    direction = 1

    # Mark positions
    for col in range(len(cipher)):
        fence[row][col] = "*"

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    # Fill ciphertext
    index = 0

    for i in range(rails):
        for j in range(len(cipher)):
            if fence[i][j] == "*":
                fence[i][j] = cipher[index]
                index += 1

    # Read zigzag
    result = ""
    row = 0
    direction = 1

    for col in range(len(cipher)):
        result += fence[row][col]

        if row == 0:
            direction = 1
        elif row == rails - 1:
            direction = -1

        row += direction

    return result

print("Encrypted:", cipher)

print("Decrypted:", plain)

import math

def inverse(a):
    for i in range(26):
        if (a * i) % 26 == 1:
            return i
    return -1


def encrypt(text, key):
    text = text.upper().replace(" ", "")

    if len(text) % 2 != 0:
        text += "X"

    result = ""

    for i in range(0, len(text), 2):
        p1 = ord(text[i]) - 65
        p2 = ord(text[i+1]) - 65

        c1 = (key[0][0]*p1 + key[0][1]*p2) % 26
        c2 = (key[1][0]*p1 + key[1][1]*p2) % 26

        result += chr(c1 + 65)
        result += chr(c2 + 65)

    return result


def decrypt(text, key):
    det = (key[0][0]*key[1][1] -
           key[0][1]*key[1][0]) % 26

    # inv = inverse(det)
    inv = pow(det,-1,26)
    print(inv)

    if inv == -1:
        return "Invalid key matrix"

    invkey = [
        [(inv * key[1][1]) % 26,
         (-inv * key[0][1]) % 26],

        [(-inv * key[1][0]) % 26,
         (inv * key[0][0]) % 26]
    ]

    result = ""

    for i in range(0, len(text), 2):
        c1 = ord(text[i]) - 65
        c2 = ord(text[i+1]) - 65

        p1 = (invkey[0][0]*c1 +
              invkey[0][1]*c2) % 26

        p2 = (invkey[1][0]*c1 +
              invkey[1][1]*c2) % 26

        result += chr(p1 + 65)
        result += chr(p2 + 65)

    return result

12. Columnar Transposition

def encrypt(text, key):
    cols = len(key)
    result = ""

    for k in sorted(key):
        col = key.index(k)

        for i in range(0, len(text), cols):
            if i + col < len(text):
                result += text[i + col]

    return result


def decrypt(cipher, key):
    cols = len(key)
    rows = (len(cipher) + cols - 1) // cols

    extra = len(cipher) % cols

    column = [""] * cols
    index = 0

    for k in sorted(key):
        col = key.index(k)

        length = rows

        if extra != 0 and col >= extra:
            length -= 1

        column[col] = cipher[index:index + length]
        index += length

    result = ""

    for i in range(rows):
        for j in range(cols):
            if i < len(column[j]):
                result += column[j][i]

    return result
